      ]
    },
    "options": []
  },
  {
    "name": "Bitrate default value",
    "description": "Sets the audio quality to 'Always High' when you first install the app.",
    "use":true,
    "compatiblePackages": {
      "com.google.android.apps.youtube.music": [
        "6.20.51",
        "6.29.59",
        "6.42.55",
        "6.51.53",
        "7.16.53",

        "8.12.54",
        "8.28.54",
        "8.30.54"

  {
    "name": "Hide ads",
